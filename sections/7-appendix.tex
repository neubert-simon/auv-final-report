\subsection{Definitions and Abbreviations}
\begin{itemize}
	\item{\textbf{Extended Stack Pointer (ESP)}}\\
	The register holding the memory address of the top of the stack \autocite{intel:dev_manual}.
	\item{\textbf{Extended Instruction Pointer (EIP)}}\\
	The register holding the return address, the address of the next instruction after a function returns \autocite{intel:dev_manual}.
	\item{\textbf{Bad characters}}\\
	In this context, \enquote{bad characters} describe all characters that terminate the processing of inputs by a system (e.g. line breaks, null bytes).
	\item{\textbf{Address Space Layout Randomization (ASLR)}}\\
	ASLR randomizes the addresses a process uses after each restart \autocite{app9142928}.
	\item{\textbf{Data Execution Prevention (DEP)}}\\
	An operating system feature that allows parts of memory to be marked as non-executable \autocite{microsoft:dep}.
	\item{\textbf{\enquote{No Operation} Opcodes (NOOPS)}}\\
	Codes that don't prompt the executing system to run any operation \autocite{arm_developer:nop}.
	\item{\textbf{\enquote{JMP ESP} instruction}}\\
	An assembly instruction, prompting the system to jump to the top of the stack to look for the next operation \autocite{arm_developer:jmp}.
	\item{\textbf{Security information and event management (SIEM) systems}}\\
	\enquote{These types of solutions collect, aggregate, and analyze large volumes of data from organization-wide applications, devices, servers, and users in real time.} \autocite{microsoft:siem}
	\item{\textbf{Intrusion detection systems (IDS)}/ \textbf{Intrusion prevention systems (IPS)}}\\
	Intrusion detection systems (IDS) are pieces of software that automatically conduct the intrusion detection process \autocite{scarfone2007guide}.
	An IPS is an IDS with the extendeded capability to attempt the prevention of detected intrusion attempts \autocite{scarfone2007guide}.
	\item{\textbf{De-Militarized Zone (DMZ)}}\\
	A De-Militarized Zone is a physical or logical network, separating trusted and untrusted parts of internal and external networks, in line with the defense-in-depth principle \autocite{8473328}.
	\itemBF{Open Source Intelligence (OSINT)}
	OSINT describes information gathering via publicly accessible sources \autocite{osint_def}.
	\itemBF{Command And Control Server (C2-Server)}
	\enquote{A Command and Control, or C\&C, server is a computer used to coordinate the actions of computers infected by a bot, rootkit, worm or other malicious software (malware) that rely on another computer for instructions and updates} \autocite{eset:C2}.
\end{itemize}
